cmake_minimum_required(VERSION 3.25)
project(OpenVpnServiceTls LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core Network RemoteObjects)
qt_standard_project_setup()
qt_add_executable(openvpn_qtro_peer main.cpp)
qt6_add_repc_replicas(openvpn_qtro_peer ../../ipc/ipc_process_interface.rep)
target_include_directories(openvpn_qtro_peer PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(openvpn_qtro_peer PRIVATE Qt6::Core Qt6::Network Qt6::RemoteObjects)
